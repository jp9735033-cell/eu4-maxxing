# eu4-maxxing: Yan on_actions for Princess Meiying life, regency, and branching missions

on_startup = {
	if = {
		limit = {
			tag = YAN
			NOT = { has_country_flag = emx_yan_divided_court_setup }
		}
		country_event = { id = emx_yan_flavor.1 }
	}
}

on_heir_death = {
	if = {
		limit = {
			tag = YAN
			NOT = { has_country_flag = emx_yan_chose_consort }
			NOT = { has_country_flag = emx_yan_event_chose_harmony }
		}
		set_country_flag = emx_yan_meiying_died
		clr_country_flag = emx_yan_meiying_is_heir
		country_event = { id = emx_yan_flavor.3 }
	}
}

on_heir_disinherited = {
	if = {
		limit = {
			tag = YAN
			NOT = { has_country_flag = emx_yan_chose_consort }
			NOT = { has_country_flag = emx_yan_event_chose_harmony }
		}
		set_country_flag = emx_yan_meiying_died
		clr_country_flag = emx_yan_meiying_is_heir
		country_event = { id = emx_yan_flavor.3 }
	}
}

on_new_monarch = {
	# When Meiying turns 15 and ascends the throne
	if = {
		limit = {
			tag = YAN
			NOT = { has_country_flag = emx_yan_meiying_died }
			is_female = yes
			has_regency = no
			has_consort_regency = no
			NOT = { has_country_flag = emx_yan_chose_consort }
			NOT = { has_country_flag = emx_yan_meiying_crowned }
		}
		country_event = { id = emx_yan_flavor.2 }
	}
}

on_bi_yearly_pulse_2 = {
	if = {
		limit = {
			tag = YAN
			NOT = { has_country_flag = emx_yan_meiying_died }
			NOT = { has_country_flag = emx_yan_chose_consort }
			NOT = { has_country_flag = emx_yan_meiying_crowned }
			has_regency = no
			has_consort_regency = no
		}
		country_event = { id = emx_yan_flavor.2 }
	}
}
