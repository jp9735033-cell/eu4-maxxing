# eu4-maxxing: shared effects for the Yan / Wu / Shu / Yue mission trees.

emx_unify_china_effect = {
	add_prestige = 25
	add_country_modifier = {
		name = emx_unified_china
		duration = -1
	}
	china_superregion = {
		limit = {
			NOT = { is_core = ROOT }
			NOT = { is_permanent_claim = ROOT }
		}
		add_permanent_claim = ROOT
	}
}

# Spice Route Scripted Effects (Strictly 1 target node's coastal CoTs per mission)

# 1. Malacca Mission -> Claims on Bengal coastal CoTs/Estuaries
emx_spice_route_malacca_effect = {
	every_province = {
		limit = {
			region = bengal_region
			has_port = yes
			OR = {
				province_has_center_of_trade_of_level = 1
				has_river_estuary_trigger = yes
			}
			NOT = { is_core = ROOT }
			NOT = { is_permanent_claim = ROOT }
		}
		add_permanent_claim = ROOT
	}
}
