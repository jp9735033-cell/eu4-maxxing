# live here once instead of being copied into four mission files.

##################################################
# Unify China
##################################################

emx_unify_china_effect = {
	add_prestige = 25
	add_country_modifier = {
		name = emx_unified_china
		duration = -1
	}
	china_superregion = {
		limit = {
			NOT = { is_core = ROOT }
			NOT = { is_permanent_claim = ROOT }
		}
		add_permanent_claim = ROOT
	}
}
