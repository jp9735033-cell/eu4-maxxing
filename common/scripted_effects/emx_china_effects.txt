# eu4-maxxing: shared effects for Chinese unification, Shu, and Yue mission trees.

emx_unify_china_effect = {
	add_prestige = 25
	add_country_modifier = {
		name = emx_unified_china
		duration = -1
	}
	# Automatically integrate Jianzhou, Manchu, or Korea if they are subject / PU
	if = {
		limit = {
			MJZ = { is_subject_of = ROOT }
		}
		inherit = MJZ
	}
	if = {
		limit = {
			MCH = { is_subject_of = ROOT }
		}
		inherit = MCH
	}
	if = {
		limit = {
			KOR = { is_subject_of = ROOT }
		}
		inherit = KOR
	}
	china_superregion = {
		limit = {
			NOT = { is_core = ROOT }
			NOT = { is_permanent_claim = ROOT }
		}
		add_permanent_claim = ROOT
	}
}

emx_add_workshop_effect = {
	if = {
		limit = { NOT = { has_building = workshop } }
		add_building = workshop
	}
	else = {
		add_base_production = 2
	}
}
