# eu4-maxxing: shared effects for the Yan / Wu / Shu / Yue mission trees.

emx_unify_china_effect = {
	add_prestige = 25
	add_country_modifier = {
		name = emx_unified_china
		duration = -1
	}
	china_superregion = {
		limit = {
			NOT = { is_core = ROOT }
			NOT = { is_permanent_claim = ROOT }
		}
		add_permanent_claim = ROOT
	}
}

# 1. Gansu Mission Effect
emx_silk_road_gansu_effect = {
	add_country_modifier = {
		name = emx_silk_road_gansu_caravan
		duration = -1
	}
	every_province = {
		limit = {
			OR = {
				area = shanshan_area
				area = kashgaria_area
				area = ferghana_area
				area = transoxiana_area
			}
			NOT = { is_core = ROOT }
			NOT = { is_permanent_claim = ROOT }
		}
		add_permanent_claim = ROOT
	}
}

# 2. Samarkand Mission Effect
emx_silk_road_samarkand_effect = {
	if = {
		limit = { has_country_modifier = emx_silk_road_gansu_caravan }
		remove_country_modifier = emx_silk_road_gansu_caravan
	}
	add_country_modifier = {
		name = emx_silk_road_samarkand_caravan
		duration = -1
	}
	every_province = {
		limit = {
			OR = {
				# Persia Route
				area = merv_area
				area = herat_area
				area = mashhad_area
				area = tabarestan_area
				area = shirvan_area
				area = tabriz_area
				# Crimea Steppe Route
				area = khiva_area
				area = nogai_area
				area = lower_yik_area
				area = astrakhan_area
				area = azov_area
				area = crimea_area
			}
			NOT = { is_core = ROOT }
			NOT = { is_permanent_claim = ROOT }
		}
		add_permanent_claim = ROOT
	}
}

		remove_country_modifier = emx_silk_road_gansu_caravan
	}

		name = emx_silk_road_crimea_caravan

	}
	change_price = {
		trade_goods = silk
		key = EMX_CRIMEA_SILK_PRICE
		value = 0.25
		duration = -1
	}
}
