# Path 1: Manchu Union Triggers
emx_yan_manchu_union_trigger = {
	custom_trigger_tooltip = {
		tooltip = emx_yan_manchu_union_trigger_tt
		has_country_flag = emx_yan_meiying_crowned
		OR = {
			junior_union_with = MJZ
			junior_union_with = JNZ
			AND = {
				exists = MJZ
				MJZ = { is_subject_of = ROOT }
			}
			AND = {
				exists = JNZ
				JNZ = { is_subject_of = ROOT }
			}
			has_country_flag = emx_yan_consort_manchu
			has_country_flag = emx_yan_chose_manchu
		}
	}
}
