# Silk Road Tiers (+5% Production Efficiency, +5% Goods Produced per active modifier)
##################################################

# Stage 1: Gansu Corridor
emx_silk_road_tier_1_gansu = {
	potential = {
		capital_scope = {
			OR = {
				region = north_china_region
				region = south_china_region
				region = xinan_region
			}
		}
	}

	trigger = {
		any_country = {
			capital_scope = {
				OR = {
					region = north_china_region
					region = south_china_region
					region = xinan_region
				}
			}
			owns_core_province = 708 # Gansu / Lanzhou
			owns_core_province = 707 # Yumen / Dunhuang
		}
	}

	production_efficiency = 0.05
	global_trade_goods_size_modifier = 0.05
}
