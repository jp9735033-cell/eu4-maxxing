# eu4-maxxing: Shu decisions.
#
# The Highland Looms, one decision per province. Each of the seventeen highland
# wool provinces - the Tibetan plateau, the Tarim oases and the Himalayan
# foothills - gets its own decision that converts that one province to cloth.
#
# Visibility: potential asks only that Shu exists and that the province still
# ships wool, which is true from 1444 whoever owns it. So all seventeen are
# listed from the first day, greyed until we own the province, and each one
# leaves the list for good once its province has been woven. Ownership is
# checked in allow so it reads as an ordinary requirement line.
#
# Cost scales with that province's development: under 10 costs 200, then 250,
# 300, 350, and 400 ducats from development 25 upwards. Every one of them
# starts in the 200 band, so weaving a province before developing it is the
# cheaper order.

country_decisions = {

	# Dege
	emx_shu_looms_2133 = {
		major = no
		potential = {
			tag = SHU
			2133 = { trade_goods = wool }
		}
		provinces_to_highlight = {
			province_id = 2133
			NOT = { owned_by = ROOT }
		}
		allow = {
			2133 = { owned_by = ROOT }
			custom_trigger_tooltip = {
				tooltip = emx_shu_loom_cost_tt
				OR = {
					AND = { 2133 = { development = 25 } treasury = 400 }
					AND = { 2133 = { development = 20 NOT = { development = 25 } } treasury = 350 }
					AND = { 2133 = { development = 15 NOT = { development = 20 } } treasury = 300 }
					AND = { 2133 = { development = 10 NOT = { development = 15 } } treasury = 250 }
					AND = { 2133 = { NOT = { development = 10 } } treasury = 200 }
				}
			}
		}
		effect = {
			if = {
				limit = { 2133 = { development = 25 } }
				add_treasury = -400
			}
			else_if = {
				limit = { 2133 = { development = 20 } }
				add_treasury = -350
			}
			else_if = {
				limit = { 2133 = { development = 15 } }
				add_treasury = -300
			}
			else_if = {
				limit = { 2133 = { development = 10 } }
				add_treasury = -250
			}
			else = {
				add_treasury = -200
			}
			2133 = { change_trade_goods = cloth }
		}
		ai_will_do = {
			factor = 1
		}
	}
