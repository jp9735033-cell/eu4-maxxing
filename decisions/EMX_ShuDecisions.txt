# eu4-maxxing: Shu decisions.
#
# The Highland Looms, one decision per province. Each of the seventeen highland
# wool provinces - the Tibetan plateau, the Tarim oases and the Himalayan
# foothills - gets its own decision that converts that one province to cloth.
#
# Visibility: potential asks only that Shu exists and that the province still
# ships wool, which is true from 1444 whoever owns it. So all seventeen are
# listed from the first day, greyed until we own the province, and each one
# leaves the list for good once its province has been woven. Ownership is
# checked in allow so it reads as an ordinary requirement line.
#
# Cost: a flat 400 ducats. The decision requires 400 in the treasury and takes
# exactly that 400 when it is passed, so it can never put us into debt. The
# game renders the requirement itself, so no custom tooltip is needed.

country_decisions = {
