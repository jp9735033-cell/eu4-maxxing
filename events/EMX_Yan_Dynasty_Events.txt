# =============================================================================
# =============================================================================
# 1. Game Start Event: The Fractured Court in Beijing
# =============================================================================
country_event = {
	id = emx_yan_flavor.1
	title = emx_yan_flavor.1.t
	desc = emx_yan_flavor.1.d
	picture = COURT_eventPicture

	fire_only_once = yes

	trigger = {
		tag = YAN
		NOT = { has_country_flag = emx_yan_divided_court_setup }
	}

	immediate = {
		set_country_flag = emx_yan_divided_court_setup
		set_country_flag = emx_origin_yan
		set_country_flag = emx_yan_meiying_is_heir
		if = { limit = { ai = no } enable_branching_mission_review_without_ai = { key = YAN } }
		set_variable = { which = yan_korea_edu value = 0 }
		set_variable = { which = yan_loyalist_edu value = 0 }
		set_variable = { which = yan_manchu_edu value = 0 }
		if = {
			limit = {
				NOT = { has_heir = yes }
			}
			define_heir = {
				name = "Meiying"
				dynasty = "Zhu"
				female = yes
				age = 5
				adm = 2
				dip = 2
				mil = 2
				claim = 100
			}
		}
		if = {
			limit = { exists = MJZ NOT = { alliance_with = MJZ } }
			create_alliance = MJZ
		}
		if = {
			limit = { exists = KOR NOT = { alliance_with = KOR } }
			create_alliance = KOR
		}
	}

	option = {
		name = emx_yan_flavor.1.a
		add_country_modifier = {
			name = "emx_yan_divided_court"
			duration = -1
		}
	}
}

# =============================================================================
# 3. Coming of Age Event: The Coronation of Queen Meiying (At Age 15 in 1454)
# Announces Meiying's coronation and sets the dynastic path!
# =============================================================================
country_event = {
	id = emx_yan_flavor.2
	title = emx_yan_flavor.2.t
	desc = emx_yan_flavor.2.d
	picture = COURT_eventPicture

	fire_only_once = yes

	trigger = {
		tag = YAN
		NOT = { has_country_flag = emx_yan_meiying_died }
		is_year = 1454
		NOT = { has_country_flag = emx_yan_chose_consort }
		NOT = { has_country_flag = emx_yan_meiying_crowned }
	}

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		set_country_flag = emx_yan_meiying_crowned
	}

	# Option A: Jurchen Steppe Betrothal (Manchu Union)
	option = {
		name = emx_yan_flavor.2.a
		trigger = {
			check_variable = { which = yan_manchu_edu which = yan_korea_edu }
			check_variable = { which = yan_manchu_edu which = yan_loyalist_edu }
		}
		set_country_flag = emx_yan_event_chose_manchu
		remove_country_modifier = emx_yan_divided_court
		add_country_modifier = {
			name = "emx_yan_manchu_consort_modifier"
			duration = -1
		}
		add_country_modifier = {
			name = "emx_yan_korean_betrayal_fallout"
			duration = 9125
		}
		if = {
			limit = { exists = MJZ }
			create_marriage = MJZ
			create_union = MJZ
			add_opinion = { who = MJZ modifier = opinion_royal_marriage }
			reverse_add_opinion = { who = MJZ modifier = opinion_royal_marriage }
			add_opinion = { who = MJZ modifier = emx_yan_chosen_dynasty_opinion }
			reverse_add_opinion = { who = MJZ modifier = emx_yan_chosen_dynasty_opinion }
		}
		if = {
			limit = { exists = KOR }
			break_alliance = KOR
			hidden_effect = {
				remove_opinion = { who = KOR modifier = opinion_royal_marriage }
				reverse_remove_opinion = { who = KOR modifier = opinion_royal_marriage }
			}
			add_opinion = { who = KOR modifier = opinion_insulted }
			reverse_add_opinion = { who = KOR modifier = opinion_insulted }
			KOR = { add_treasury = 500 }
		}
		add_treasury = -500
		1816 = {
			noble_rebels = 2
		}
		add_prestige = 15
		add_legitimacy = 20
	}

	# Option B: Joseon Scholarly Marriage (Korean Union)
	option = {
		name = emx_yan_flavor.2.b
		trigger = {
			check_variable = { which = yan_korea_edu which = yan_manchu_edu }
			check_variable = { which = yan_korea_edu which = yan_loyalist_edu }
		}
		set_country_flag = emx_yan_event_chose_korea
		remove_country_modifier = emx_yan_divided_court
		add_country_modifier = {
			name = "emx_yan_korean_consort_modifier"
			duration = -1
		}
		if = {
			limit = { exists = KOR }
			break_alliance = KOR
			create_marriage = KOR
			create_union = KOR
			add_opinion = { who = KOR modifier = opinion_royal_marriage }
			reverse_add_opinion = { who = KOR modifier = opinion_royal_marriage }
			add_opinion = { who = KOR modifier = emx_yan_chosen_dynasty_opinion }
			reverse_add_opinion = { who = KOR modifier = emx_yan_chosen_dynasty_opinion }
		}
		liaoning_area = {
			limit = {
				NOT = { is_core = ROOT }
				NOT = { is_permanent_claim = ROOT }
			}
			add_permanent_claim = ROOT
		}
		if = {
			limit = { exists = MJZ }
			break_alliance = MJZ
			hidden_effect = {
				remove_opinion = { who = MJZ modifier = opinion_royal_marriage }
				reverse_remove_opinion = { who = MJZ modifier = opinion_royal_marriage }
				remove_truce = MJZ
				MJZ = { remove_truce = ROOT }
			}
			add_opinion = { who = MJZ modifier = opinion_insulted }
			reverse_add_opinion = { who = MJZ modifier = opinion_insulted }
			MJZ = {
				declare_war_with_cb = {
					who = ROOT
					casus_belli = cb_insult
				}
			}
		}
		add_prestige = 15
		add_legitimacy = 20
	}

	# Option C: Imperial Autonomy & Balance (Harmony Path)
	option = {
		name = emx_yan_flavor.2.c
		trigger = {
			check_variable = { which = yan_loyalist_edu which = yan_manchu_edu }
			check_variable = { which = yan_loyalist_edu which = yan_korea_edu }
		}
		set_country_flag = emx_yan_event_chose_harmony
		remove_country_modifier = emx_yan_divided_court
		add_country_modifier = {
			name = "emx_yan_harmonized_court"
			duration = -1
		}
		if = {
			limit = { exists = MJZ }
			break_alliance = MJZ
			hidden_effect = {
				remove_opinion = { who = MJZ modifier = opinion_royal_marriage }
				reverse_remove_opinion = { who = MJZ modifier = opinion_royal_marriage }
			}
		}
		if = {
			limit = { exists = KOR }
			break_alliance = KOR
			hidden_effect = {
				remove_opinion = { who = KOR modifier = opinion_royal_marriage }
				reverse_remove_opinion = { who = KOR modifier = opinion_royal_marriage }
			}
		}
		add_prestige = 25
		add_legitimacy = 25
	}
}
